cmake_minimum_required(VERSION 3.16)
project(UsefulCppCollection LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The combined example build currently targets Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(cout-to-file "Cout to File/Project2/main.cpp")
add_executable(cursor-replay "Cursor Capture and Replay/Next/main.cpp")
add_executable(cursor-surveillance "Cursor_Surveilance/Cursor_Surveilance/main.cpp"
    "Cursor_Surveilance/Cursor_Surveilance/Cursor_Surveillance.cpp")
add_executable(hide-cursor "Hide Windows Cursor and bring it Back/Project1/maincpp.cpp")
add_executable(command-buffer "ISystemCommandCmdBuffer/ISystemCommandCmdBuffer/Sources/main.cpp"
    "ISystemCommandCmdBuffer/ISystemCommandCmdBuffer/Sources/SystemCommandOutputBuffer.cpp")
add_executable(key-sequence "KeyToggle KeySequenceBuffer/KeyToggle/main.cpp")
add_executable(filesystem-scan "Multithreaded Filesystem Operator/Project2/main.cpp"
    "Multithreaded Filesystem Operator/Project2/scan.cpp")
add_executable(pipe-capture "Popen/Popen/main.cpp")
add_executable(aes-demo "Test AES/Test AES/main.cpp" "Test AES/Test AES/AES.cpp")
add_executable(mouse-hook "Windows Cursor Hook/TESTLL/main.cpp")
add_executable(scanf-demo "sscanf parse/Project3/Quelle.cpp")
foreach(target cursor-replay cursor-surveillance hide-cursor key-sequence mouse-hook)
    target_link_libraries(${target} PRIVATE user32)
endforeach()
target_link_libraries(filesystem-scan PRIVATE Threads::Threads)
if(MINGW)
    target_link_options(filesystem-scan PRIVATE -municode)
endif()
target_link_libraries(mouse-hook PRIVATE Threads::Threads)
foreach(target cout-to-file cursor-replay cursor-surveillance hide-cursor command-buffer
        key-sequence filesystem-scan pipe-capture mouse-hook scanf-demo)
    if(MSVC)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME bounded-scanf COMMAND scanf-demo)
    set_tests_properties(bounded-scanf PROPERTIES PASS_REGULAR_EXPRESSION "Rudolph -> 12")
    add_test(NAME pipe-output COMMAND pipe-capture)
    set_tests_properties(pipe-output PROPERTIES PASS_REGULAR_EXPRESSION "Pipe capture demo")
    add_test(NAME key-sequence-state COMMAND key-sequence --self-test)
    add_test(NAME complete-aes-block COMMAND aes-demo)
    add_test(NAME restored-console-buffer COMMAND cout-to-file "${CMAKE_CURRENT_BINARY_DIR}/cout-test.txt")
    set_tests_properties(restored-console-buffer PROPERTIES PASS_REGULAR_EXPRESSION "Console output restored")
    add_test(NAME command-output COMMAND command-buffer "${CMAKE_CURRENT_BINARY_DIR}/command-test.txt")
    set_tests_properties(command-output PROPERTIES PASS_REGULAR_EXPRESSION "System command output")
endif()
